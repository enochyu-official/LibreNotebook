\section{Practice Problems}

\begin{enumerate}
\item
Determine whether a transformation defined as $T(\langle a, b \rangle)
= a + b$ is linear. How about $T(\langle a, b \rangle) = ab$?

\item
For square matrices $A$ and $B$ of order $n \times n$, show that if
$\lambda$ is an eigenvalue of $AB$, then the eigenvalues of $BA$
also include $\lambda$ \cite{MATH02-3}.

\item
Find bases for the kernel and the image of
$\begin{bmatrix}
    1 & 1 & 1 \\
    0 & 1 & 0
\end{bmatrix}$.

\item
Show that $A \cong B$ if and only if $A^\intercal \cong B^\intercal$.

\item
Find the eigenvalues and corresponding eigenspace for
$\begin{bmatrix}
    1 & 2 & 3 \\
    0 & 1 & 0 \\
    0 & 1 & 2
\end{bmatrix}$.
\end{enumerate}
